% 2 MOTIVATION (concept)
\slidefb{Where should we stimulate?}{concept_motivation}{%
  \bul{%
    \item Alzheimer's disrupts connectivity \textbf{brain-wide}
    \item Neuromodulation works best at a \textbf{single site}
    \item Is the signature reducible to a \textcolor{teal}{focal target}?}}

% 3 DIGITAL TWIN (concept)
\slidefb{A perturbable digital twin}{concept_twin}{%
  \bul{%
    \item One personalised model \textbf{per patient}
    \item Its free-running dynamics \textbf{reproduce} the patient's own
    \item We can now intervene \textcolor{teal}{in silico}}}

% 4 MODEL FAITHFUL (paper fig 2)
\slidefb{The model is faithful and individual}{figure2_model}{%
  \bul{%
    \item Reconstructs each subject's connectivity
    \item Identifiable \textbf{across} subjects
    \item A generative surrogate we can \textcolor{teal}{modify}}}

% 5 CLASSIFICATION (paper fig 3)
\slidefb{Reading the disease from the model}{figure3_classification}{%
  \bul{%
    \item Model parameters separate \textcolor{ctrl}{controls} from \textcolor{adred}{AD}
    \item Modest accuracy --- below structural atrophy
    \item A \textcolor{teal}{target} for stimulation, not a diagnostic}}

% 6 TWO INTERVENTIONS (concept)
\slidefb{Two ways to intervene}{concept_interventions}{%
  \bul{%
    \item \textbf{Graph}: reshape the connectivity itself
    \item \textbf{Activity}: a focal drive at one node
    \item They need not point to the \textcolor{teal}{same target}}}

% 7 RESULT 1 (paper fig 4)
\slidefb{Correcting connectivity reverts the signature}{figure4_stimulation}{%
  \bul{%
    \item Shifting the \textbf{whole} graph $\rightarrow$ reclassified as control
    \item \textcolor{adred}{Focal edits fail} at any amplitude
    \item The correction is \textcolor{teal}{distributed}}}

% 8 RESULT 2 (paper fig 5 -> topsites)
\slidefb{The best target is not the most affected site}{figure5_topsites}{%
  \bul{%
    \item \textcolor{adred}{Most-changed} sites: deep, subcortical
    \item \textcolor{teal}{Effective} targets: cortical, well-connected
    \item The two sets are \textbf{largely disjoint}}}

% 9 WHY (concept)
\slidefb{It follows from the connectivity}{concept_hubs}{%
  \bul{%
    \item \textcolor{adred}{Most-affected} nodes are weakly connected
    \item Weak nodes are \textbf{poor actuators}
    \item \textcolor{teal}{Hubs} spread the correction network-wide}}

% 10 RESULT 3 (paper fig 6 -> singlecompare)
\slidefb{One site, and closed-loop control}{figure6_singlecompare}{%
  \bul{%
    \item Resonant drive at the \textcolor{teal}{right site} cures the patient
    \item A single read-out edit is \textbf{inert}
    \item Closed-loop titration $\rightarrow$ \textbf{lower dose}}}
